# dictz bash completion — add to ~/.bashrc:  eval "$(dictz completion bash)"
_dictz() {
    local cur prev
    cur="${COMP_WORDS[COMP_CWORD]}"
    prev="${COMP_WORDS[1]}"
    if [ "$COMP_CWORD" -eq 1 ]; then
        COMPREPLY=( $(compgen -W "add catalog completion config dicts disable doctor downloads dump enable f fav favorites fuzzy gui help history info install l licenses lookup ls move paths popup random reindex remove rm s scan selftest setup shell shortcut show sources suggest uninstall update upgrade version watch --version --help" -- "$cur") )
        return
    fi
    case "$prev" in
        enable|disable|remove|rm|move|info|reindex|dump)
            local IFS=$'\n'
            COMPREPLY=( $(compgen -W "$(dictz dicts --names 2>/dev/null)" -- "$cur") ) ;;
        config) COMPREPLY=( $(compgen -W "list get set reset path export import" -- "$cur") ) ;;
        favorites|fav) COMPREPLY=( $(compgen -W "list add remove clear" -- "$cur") ) ;;
        completion) COMPREPLY=( $(compgen -W "bash zsh fish powershell" -- "$cur") ) ;;
        watch) COMPREPLY=( $(compgen -W "on off toggle status" -- "$cur") ) ;;
        shortcut) COMPREPLY=( $(compgen -W "--remove --desktop-only --menu-only" -- "$cur") ) ;;
        sources) COMPREPLY=( $(compgen -W "list add edit remove enable disable default up down refresh export import reset" -- "$cur") ) ;;
        catalog) COMPREPLY=( $(compgen -W "search show languages updates build check" -- "$cur") ) ;;
        downloads) COMPREPLY=( $(compgen -W "list clear" -- "$cur") ) ;;
        setup) COMPREPLY=( $(compgen -W "export import show" -- "$cur") ) ;;
        add|scan) COMPREPLY=( $(compgen -f -- "$cur") ) ;;
    esac
}
complete -o default -F _dictz dictz dct
