# dictz PowerShell completion — add to $PROFILE:  dictz completion powershell | Out-String | Invoke-Expression
Register-ArgumentCompleter -Native -CommandName dictz -ScriptBlock {
    param($wordToComplete, $commandAst, $cursorPosition)
    $words = $commandAst.CommandElements | ForEach-Object { $_.ToString() }
    if ($words.Count -le 2) {
        @('add', 'completion', 'config', 'dicts', 'disable', 'doctor', 'dump', 'enable', 'f', 'fav', 'favorites', 'fuzzy', 'gui', 'help', 'history', 'info', 'l', 'lookup', 'ls', 'move', 'paths', 'popup', 'random', 'reindex', 'remove', 'rm', 's', 'scan', 'selftest', 'shell', 'show', 'suggest', 'update', 'upgrade', 'version', 'watch') | Where-Object { $_ -like "$wordToComplete*" } |
            ForEach-Object { [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_) }
        return
    }
    switch ($words[1]) {
        { $_ -in 'enable','disable','remove','rm','move','info','reindex','dump' } {
            dictz dicts --names | Where-Object { $_ -like "$wordToComplete*" } |
                ForEach-Object { [System.Management.Automation.CompletionResult]::new("'$_'", $_, 'ParameterValue', $_) }
        }
        'config' { @('list','get','set','reset','path','export','import') | Where-Object { $_ -like "$wordToComplete*" } }
        'completion' { @('bash','zsh','fish','powershell') | Where-Object { $_ -like "$wordToComplete*" } }
        'watch' { @('on','off','toggle','status') | Where-Object { $_ -like "$wordToComplete*" } }
    }
}
